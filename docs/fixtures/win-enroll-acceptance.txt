# win-enroll-acceptance.txt - Windows acceptance harness for the enroll precheck on the
# user's PC (real Windows PowerShell 5.1). Use only after a packed build reviewed outside
# this file is installed and its hashes have been injected; authorization for running it
# is established outside this file. NOT for general use.
# It extracts the EXACT shipped C# and helper block from win-enroll-precheck.txt (single
# source - no copies) and exercises them end-to-end against real processes:
#   - a locally compiled test-double exe (Add-Type, no downloads) for timeout, descendant,
#     fault-injection (create/assign/resume/query), parser and closure scenarios
#   - the real installed app, launched ONLY after the injected pins verify exactly
# Task Scheduler: this harness performs NO scheduler mutations - no create, delete,
# replace, or /f anywhere. A read-only query reports whether GlassWidgets-Watchdog
# exists; removing it is a separate owner step (win-watchdog-remove.txt) run by the
# owner outside this harness, never from here.
# Every run is tracked with an UNKNOWN placeholder before invocation; the staging dir is
# removed only when closure is proven for ALL runs, and preserved otherwise. Job closure
# (kernel-verified job emptiness) is the containment proof - a process-name query is only
# a supporting signal, never proof of child absence.
# Every precheck invocation is gated: run assessment (launcher ok, started, no timeout,
# proven job closure, exit 0, no surviving descendants) and an UNTRUNCATED capture come
# BEFORE any stdout text is read; timeout, nonzero exit, or partial output is never a PASS.
param([string]$AsarSha256='',[string]$ExeSha256='',[string]$ExpectedVersion='',[string]$PrecheckPath='')
$ErrorActionPreference='Continue'
$ProgressPreference='SilentlyContinue'
function Say($m){ Write-Host $m }
$script:pass=0; $script:fail=0; $script:skip=0
function Scenario($name,[scriptblock]$fn){
  try{ $ok=& $fn }catch{ Say ('FAIL '+$name+' (exception: '+$_.Exception.Message+')'); $script:fail++; return }
  if($ok -eq $true){ Say ('PASS '+$name); $script:pass++ }
  elseif($ok -eq 'SKIP'){ Say ('SKIP '+$name); $script:skip++ }
  else{ Say ('FAIL '+$name); $script:fail++ }
}
Say '== GlassWidgets Windows acceptance (real PS) =='
Say ('PS version: '+$PSVersionTable.PSVersion.ToString())
Scenario 'ps-is-real-5.1' { ($PSVersionTable.PSVersion.Major -eq 5) }

$stage=Join-Path $env:TEMP ('gw-accept-'+[guid]::NewGuid().ToString('N'))
New-Item -ItemType Directory -Force -Path $stage | Out-Null
$appExe=Join-Path $env:LOCALAPPDATA 'Programs\GlassWidgets\GlassWidgets.exe'
$appHome=Join-Path $env:LOCALAPPDATA 'Programs\GlassWidgets'
$psExe=Join-Path $env:SystemRoot 'System32\WindowsPowerShell\v1.0\powershell.exe'
if(-not $PrecheckPath){ $PrecheckPath=Join-Path $PSScriptRoot 'win-enroll-precheck.txt' }
$doubleExe=Join-Path $stage 'gw-double.exe'
$outDir=Join-Path $stage 'caps'; New-Item -ItemType Directory -Force -Path $outDir | Out-Null
$script:runs=@()
function Run-Tracked {
  # UNKNOWN placeholder recorded BEFORE invocation; replaced by the real result. If the
  # call escapes, the placeholder forces stage preservation (doubt never cleans up).
  param([scriptblock]$Call)
  $script:runs+=[pscustomobject]@{started=$null;treeCleaned=$false}
  $r=& $Call
  $script:runs[$script:runs.Count-1]=$r
  return $r
}
$pinsReady=($AsarSha256 -and $ExeSha256 -and $ExpectedVersion)
function Test-RealAppPins {
  # '' only when the installed app matches the injected reviewed pins EXACTLY - the gate
  # for ANY real-app invocation in this harness. Never launches on mismatch or doubt.
  if(-not $pinsReady){ return 'pins not injected' }
  if(-not (Test-Path $appExe)){ return 'installed exe not found' }
  $asar=Join-Path $appHome 'resources\app.asar'
  if(-not (Test-Path $asar)){ return 'app.asar not found' }
  $hx=(Get-FileHash -LiteralPath $appExe -Algorithm SHA256).Hash
  $ha=(Get-FileHash -LiteralPath $asar -Algorithm SHA256).Hash
  if($hx -ine $ExeSha256){ return ('installed exe hash does not match the injected reviewed exe pin') }
  if($ha -ine $AsarSha256){ return ('installed asar hash does not match the injected reviewed asar pin') }
  $vi=(Get-Item $appExe).VersionInfo
  $verText=[string]$vi.ProductVersion; if([string]::IsNullOrWhiteSpace($verText)){ $verText=[string]$vi.FileVersion }
  $vg=Test-VersionGate -VerText $verText -ExpectedVersion $ExpectedVersion
  if($vg){ return $vg }
  return ''
}

$precheckText=[System.IO.File]::ReadAllText($PrecheckPath)
$csMark=$precheckText.IndexOf('# GW-CS-BEGIN')
$csStart=$precheckText.IndexOf("@'",$csMark)+2
$csEnd=$precheckText.IndexOf("'@",$csStart)
$hs=$precheckText.IndexOf('function ConvertTo-NormalizedVersion')
$hm=$precheckText.IndexOf('# ---------- main flow ----------')
$compiledAll=$false
if($csMark -lt 0 -or $csStart -lt 2 -or $csEnd -le $csStart -or $hs -lt 0 -or $hm -le $hs){ Say 'FATAL: cannot extract shipped blocks from the precheck file' }
else{
  try{
    Add-Type -TypeDefinition $precheckText.Substring($csStart,$csEnd-$csStart) -ErrorAction Stop
    . ([ScriptBlock]::Create($precheckText.Substring($hs,$hm-$hs)))
    Add-Type -TypeDefinition $doubleCs -OutputAssembly $doubleExe -OutputType ConsoleApplication -ErrorAction Stop
    $compiledAll=$true
  }catch{ Say ('FATAL: local compile failed: '+$_.Exception.Message) }
}

# supporting signal only - a name query is NOT proof of child absence; job closure is
function No-Leftovers { @(Get-Process -Name 'gw-double' -ErrorAction SilentlyContinue).Count -eq 0 }
function Report-Closure($r){ Say ('  closure: treeCleaned='+$r.treeCleaned+' activeAfterExit='+$r.activeAfterExit+' timedOut='+$r.timedOut+' exit='+$r.exitCode) }
function Run-Double {
  param([string[]]$Args2,[int]$TimeoutMs=20000,[int]$Fault=0)
  return Run-Tracked {
    Invoke-BoundedProcess -ExePath $doubleExe -Arguments $Args2 -OutDir $outDir -TimeoutMs $TimeoutMs -Runner {
      param($e,$a,$op,$ep,$t)
      $r=[GwBounded]::Run($e,[string[]]$a,$op,$ep,$t,[string[]]@(),[string[]]@(),$Fault)
      [pscustomobject]@{started=[bool]$r.Started;timedOut=[bool]$r.TimedOut;treeCleaned=[bool]$r.ClosureProven;activeAfterExit=[int]$r.ActiveAfterExit;exitCode=[int]$r.ExitCode;apiError=[string]$r.Error}
    }.GetNewClosure()
  }
}

# --- precheck invocations: real PS 5.1 child, BOUNDED (job-contained, 60s outer timeout) ---
# The child's stdin is the NUL device, so the precheck's closing Read-Host gets an
# immediate EOF and can never hang the harness; stdout is captured capped via the shipped
# bounded reader.
function Invoke-Precheck {
  param([hashtable]$Params)
  $argList=@('-NoProfile','-ExecutionPolicy','Bypass','-File',$PrecheckPath)
  foreach($k in $Params.Keys){ $argList+=('-'+$k); $argList+=([string]$Params[$k]) }
  Run-Tracked { Invoke-BoundedProcess -ExePath $psExe -Arguments $argList -OutDir $outDir -TimeoutMs 60000 }
}
function Assert-RunUsable($r,[string]$label){
  # '' only when the run may be judged by its stdout: launcher ok, started, no timeout,
  # proven job closure, exit 0, no surviving descendants, and the capture is NOT
  # truncated. Every other state is an explicit rejection - partial, timed-out, or
  # nonzero-exit text is never evidence, and never a PASS.
  $ra=Get-RunAssessment -Run $r -Label $label
  if($ra){ return $ra }
  if($r.overCap){ return ($label+' capture exceeded the 64KB cap - truncated output is never evidence; rejected') }
  return ''
}

Scenario 'real-app-bad-nonce-exits-nonzero' {
  # gated: the real app is launched ONLY after the injected pins verify exactly
  if(-not $compiledAll){ return 'SKIP' }
  $gate=Test-RealAppPins
  if($gate -eq 'pins not injected'){ return 'SKIP' }
  if($gate){ Say ('  gate: '+$gate); return $false }
  $r=Run-Tracked { Invoke-BoundedProcess -ExePath $appExe -Arguments @('--gw-runtime-paths-json','x') -OutDir $outDir -EnvRemove @{ELECTRON_RUN_AS_NODE=1} -TimeoutMs 20000 }
  # exact adjudication, not any rejection: the app must have exited with precisely code 2
  # (bad nonce), inside a proven-empty job, with no timeout and no launcher error, and its
  # stdout must contain NO JSON evidence line. Any other failure shape is UNVERIFIED.
  Report-Closure $r
  if([string]$r.apiError){ Say ('  UNVERIFIED: launcher error: '+[string]$r.apiError); return $false }
  if($r.timedOut){ Say '  UNVERIFIED: bad-nonce run timed out'; return $false }
  if($r.treeCleaned -ne $true){ Say '  UNVERIFIED: closure not proven for the bad-nonce run'; return $false }
  if($r.exitCode -ne 2){ Say ('  UNVERIFIED: expected exit code 2 for a bad nonce, got '+$r.exitCode); return $false }
  if([string]$r.stdout -match '"v":1'){ Say '  UNVERIFIED: evidence JSON was emitted despite the bad nonce'; return $false }
  $true
}

Scenario 'installer-shortcut-renamed-old-absent' {
  if(-not (Test-Path $appExe)){ return 'SKIP' }
  $newName=-join [char[]](0x05D7,0x05D9,0x05D9,0x05DD,0x20,0x05E9,0x05DC,0x05D9)
  $sm=Join-Path $env:APPDATA 'Microsoft\Windows\Start Menu\Programs'
  $desk=[Environment]::GetFolderPath('Desktop')
  $ok=$true
  # the OLD name must be absent everywhere it could appear: the Start Menu folder, the
  # Start Menu ROOT, and the Desktop
  foreach($c in @((Join-Path $sm 'GlassWidgets\GlassWidgets.lnk'),(Join-Path $sm 'GlassWidgets.lnk'),(Join-Path $desk 'GlassWidgets.lnk'))){
    if(Test-Path -LiteralPath $c){ Say ('  UNVERIFIED: old shortcut still present: '+$c); $ok=$false }
  }
  # the RENAMED shortcut is required in BOTH the Start Menu (folder or root) AND the
  # Desktop, and every found one must target the installed exe exactly
  $smNew=@()
  foreach($c in @((Join-Path $sm ('GlassWidgets\'+$newName+'.lnk')),(Join-Path $sm ($newName+'.lnk')))){
    if(Test-Path -LiteralPath $c){ $smNew+=$c }
  }
  $deskNew=Join-Path $desk ($newName+'.lnk')
  if($smNew.Count -eq 0){ Say '  UNVERIFIED: no renamed shortcut in the Start Menu (folder or root)'; $ok=$false }
  if(-not (Test-Path -LiteralPath $deskNew)){ Say '  UNVERIFIED: no renamed shortcut on the Desktop'; $ok=$false }
  foreach($c in (@($smNew)+@($deskNew))){
    if(-not (Test-Path -LiteralPath $c)){ continue }
    try{
      $sh=(New-Object -ComObject WScript.Shell).CreateShortcut($c)
      if([string]$sh.TargetPath -ine $appExe){ Say ('  UNVERIFIED: shortcut target is not the installed exe: '+$c+' -> '+[string]$sh.TargetPath); $ok=$false }
      else{ Say ('  shortcut ok: '+$c+' -> '+[string]$sh.TargetPath) }
    }catch{ Say ('  UNVERIFIED: cannot read shortcut '+$c+': '+$_.Exception.Message); $ok=$false }
  }
  $ok
}

# --- Task Scheduler: READ-ONLY state only. This harness performs NO scheduler
# mutations anywhere - no create, no delete, no replace, no /f, no SYSTEM task. A
# dedicated task in a test namespace was considered and intentionally skipped: the PC
# sees zero task mutations from this harness. Removing GlassWidgets-Watchdog (when the
# owner wants it gone) is a separate owner step via win-watchdog-remove.txt, run by the
# owner outside this harness - never invoked, created, or replaced from here.
function Test-WdPresent {
  # tri-state: $true/$false only on definitive answers, $null (UNVERIFIED) otherwise.
  # ABSENT is proven ONLY by GetTask itself throwing file-not-found (0x80070002); a
  # failure in Connect or GetFolder proves nothing about the task and stays UNVERIFIED.
  $c=$null; $f=$null
  try{ $c=New-Object -ComObject 'Schedule.Service' }catch{ return $null }
  try{ $c.Connect() }catch{ return $null }
  try{ $f=$c.GetFolder('\') }catch{ return $null }
  try{ $null=$f.GetTask('GlassWidgets-Watchdog'); return $true }
  catch [System.IO.FileNotFoundException]{ if($_.Exception.HResult -eq -2147024894){ return $false }; return $null }
  catch [System.Runtime.InteropServices.COMException]{ if($_.Exception.HResult -eq -2147024894){ return $false }; return $null }
  catch{ return $null }
}
Scenario 'watchdog-state-readonly-query' {
  $s=Test-WdPresent
  if($null -eq $s){ Say '  UNVERIFIED: scheduler query failed - task state unknown'; return $false }
  if($s){ Say '  state: GlassWidgets-Watchdog PRESENT - the separate owner removal step still applies' }
  else{ Say '  state: GlassWidgets-Watchdog absent' }
  $true
}
