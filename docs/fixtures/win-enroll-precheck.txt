# win-enroll-precheck.txt - run on the user's PC BEFORE any enroll send (regular PowerShell)
# READ-ONLY: changes nothing, prints no key material.
# Why: the installed 0.4.63 (8288dce) collapses a missing / corrupt / ACL-unverifiable device key
# into "no key", and a new enroll can then REPLACE device.json. No-rotation is only guaranteed
# for a healthy, successfully loaded identity. This tells us which case this PC is in:
#   VERDICT ABSENT   - no identity yet: an enroll will mint a new one (expected for first time)
#   VERDICT HEALTHY  - valid identity (parses, public key derives from private key, owner-only ACL):
#                      an enroll returns the existing identity, nothing is rewritten
#   VERDICT UNHEALTHY / UNREADABLE - a file is there but not provably healthy: an enroll could
#                      REPLACE it. DO NOT send an enroll; send this output back instead.
$ErrorActionPreference='Stop'
$ProgressPreference='SilentlyContinue'
function Say($m){ Write-Host $m }
$d=[Environment]::GetFolderPath('Desktop')
try{ Start-Transcript -Path (Join-Path $d 'enroll-precheck-log.txt') -Force | Out-Null }catch{}
$appExe=Join-Path $env:LOCALAPPDATA 'Programs\GlassWidgets\GlassWidgets.exe'
$devDir=Join-Path $env:APPDATA 'glasswidgets\device'
$devFile=Join-Path $devDir 'device.json'
$probe=Join-Path $env:TEMP 'gw-precheck-probe.js'
$probeOut=Join-Path $env:TEMP 'gw-precheck-out.txt'
$verdict='UNKNOWN'; $reasons=@()
try{
  Say '== GlassWidgets enroll precheck (read-only) =='
  if(Test-Path $appExe){
    $vi=(Get-Item $appExe).VersionInfo
    Say ('app version: '+$vi.ProductVersion)
  }else{ $reasons+='installed exe not found at '+$appExe }
  # 1. strict presence: only "not found" counts as absent; any other failure is UNREADABLE
  $present=$false; $unreadable=$false
  try{ $null=Get-Item -LiteralPath $devFile -Force -ErrorAction Stop; $present=$true }
  catch [System.Management.Automation.ItemNotFoundException]{ $present=$false }
  catch [System.IO.DirectoryNotFoundException]{ $present=$false }
  catch [System.IO.FileNotFoundException]{ $present=$false }
  catch{ $unreadable=$true; $reasons+='cannot inspect device.json: '+$_.Exception.GetType().Name }
  if($unreadable){ $verdict='UNREADABLE' }
  elseif(-not $present){
    $verdict='ABSENT'
    if(Test-Path $devDir){ $reasons+='device directory exists but device.json does not' }
  }else{
    # 2. ACL owner-only on the file AND the directory (exact current-user SID)
    $mySid=[System.Security.Principal.WindowsIdentity]::GetCurrent().User.Value
    function Test-OwnerOnly($p,$isDir){
      try{
        $a=Get-Acl -LiteralPath $p; $rules=@($a.Access)
        if(-not $a.AreAccessRulesProtected){ return 'inheritance on: '+$p }
        if($rules.Count -ne 1){ return 'rule count '+$rules.Count+' on '+$p }
        $sid=$rules[0].IdentityReference.Translate([System.Security.Principal.SecurityIdentifier]).Value
        if($sid -ne $mySid){ return 'rule SID is not the current user on '+$p }
        if($rules[0].AccessControlType -ne 'Allow'){ return 'rule is not Allow on '+$p }
        $rw=[string]$rules[0].FileSystemRights
        if($isDir){ if($rw -notmatch 'FullControl'){ return 'dir rights not FullControl: '+$rw } }
        elseif($rw -notmatch 'FullControl' -and ($rw -notmatch 'Read' -or $rw -notmatch 'Write')){ return 'file rights missing Read/Write: '+$rw }
        return ''
      }catch{ return 'ACL check failed: '+$_.Exception.Message }
    }
    $aclF=Test-OwnerOnly $devFile $false
    $aclD=Test-OwnerOnly $devDir $true
    if($aclF){ $reasons+=('file ACL: '+$aclF) }
    if($aclD){ $reasons+=('dir ACL: '+$aclD) }
    # 3. content health, checked with the app's OWN runtime (same node as the app, no downloads)
    $contentOk=$false
    if(-not (Test-Path $appExe)){ $reasons+='cannot check key content: app exe missing' }
    else{
      $js=@'
const fs=require('fs'),crypto=require('crypto');
try{
  const j=JSON.parse(fs.readFileSync(process.argv[2],'utf8'));
  const priv=crypto.createPrivateKey({key:Buffer.from(String(j.priv),'base64'),format:'der',type:'pkcs8'});
  const pub=crypto.createPublicKey(priv).export({format:'der',type:'spki'}).toString('base64');
  if(pub!==String(j.pub)){console.log('MISMATCH');}
  else{console.log('OK '+crypto.createHash('sha256').update(pub).digest('hex').slice(0,16));}
}catch(e){console.log('CORRUPT');}
'@
      Set-Content -Path $probe -Value $js -Encoding ASCII
      if(Test-Path $probeOut){ Remove-Item $probeOut -Force }
      $prevEap=$ErrorActionPreference; $ErrorActionPreference='Continue'
      try{
        & cmd.exe /c ('set ELECTRON_RUN_AS_NODE=1&& "'+$appExe+'" "'+$probe+'" "'+$devFile+'" >"'+$probeOut+'" 2>&1') | Out-Null
      }finally{ $ErrorActionPreference=$prevEap }
      $res=''; if(Test-Path $probeOut){ $res=(Get-Content $probeOut -Raw).Trim() }
      if($res -match '^OK ([0-9a-f]{16})$'){ $contentOk=$true; $devId=$Matches[1] }
      elseif($res -eq 'MISMATCH'){ $reasons+='stored public key does NOT derive from the private key' }
      elseif($res -eq 'CORRUPT'){ $reasons+='device.json is not valid (unparseable / missing fields / bad key)' }
      else{ $reasons+='could not run the content probe (got: '+$res+')' }
    }
    if($contentOk -and -not $aclF -and -not $aclD){ $verdict='HEALTHY' }
    else{ $verdict='UNHEALTHY' }
  }
  Say ''
  Say ('VERDICT: '+$verdict)
  if($verdict -eq 'HEALTHY'){ Say ('deviceId (derived): '+$devId); Say 'meaning: an enroll returns this existing identity; nothing is rewritten.' }
  if($verdict -eq 'ABSENT'){ Say 'meaning: no identity yet - an enroll will mint a new one (expected for a first-time pairing).' }
  if($verdict -eq 'UNHEALTHY' -or $verdict -eq 'UNREADABLE'){ Say 'meaning: DO NOT send an enroll - the installed version could REPLACE this file. Send this output back.' }
  foreach($r in $reasons){ Say ('  - '+$r) }
  Say ''
  Say 'This script changed nothing. Send this output back (and enroll-precheck-log.txt from the Desktop).'
}catch{
  Say ('FAIL: '+$_.Exception.Message)
  Say 'Send this whole output back.'
}finally{
  Remove-Item $probe -Force -ErrorAction SilentlyContinue
  Remove-Item $probeOut -Force -ErrorAction SilentlyContinue
  try{ Stop-Transcript | Out-Null }catch{}
  Read-Host 'Press Enter to close'
}
