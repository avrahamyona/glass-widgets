# win-enroll-precheck.txt - run on the user's PC (regular PowerShell, no admin needed)
# DIAGNOSTIC ONLY. This script gathers evidence and findings; it draws NO enroll conclusions,
# issues NO permits, and changes NO app identity or settings. It writes a Desktop transcript
# (enroll-precheck-log.txt) and a unique GUID temp staging dir (removed at the end). Launching
# the app's evidence mode may cause Electron's own cache/log writes - zero writes is NOT claimed.
#
# What it does:
#   1. Runs the installed app's --gw-runtime-paths-json <nonce> mode (0.4.64+) - the SINGLE
#      agreed way to learn the packed runtime's OWN userData root. The nonce, isPackaged, exe
#      path and version are verified; a mismatch means the evidence is rejected.
#   2. Reports FILE_FINDING for device.json at the returned root: healthy / absent / unhealthy /
#      unreadable / not_checked. On an older app without the mode, or on any evidence failure,
#      VERDICT stays UNKNOWN and the file is NOT checked against any guessed root.
# ACL comparison mirrors the app's shared evaluator (src/lib/acl-check.js) exactly.
param([string]$AsarSha256='') # optional pin: sha256 of resources\app.asar from the REVIEWED 0.4.64 build
$ErrorActionPreference='Stop'
$ProgressPreference='SilentlyContinue'
function Say($m){ Write-Host $m }
$d=[Environment]::GetFolderPath('Desktop')
try{ Start-Transcript -Path (Join-Path $d 'enroll-precheck-log.txt') -Force | Out-Null }catch{}
$appExe=Join-Path $env:LOCALAPPDATA 'Programs\GlassWidgets\GlassWidgets.exe'
$stage=Join-Path $env:TEMP ('gw-precheck-'+[guid]::NewGuid().ToString('N'))
New-Item -ItemType Directory -Force -Path $stage | Out-Null
$probe=Join-Path $stage 'probe.js'
$probeOut=Join-Path $stage 'out.txt'
$cap=Join-Path $stage 'rt.json'
$verdict='UNKNOWN'; $finding='not_checked'; $reasons=@(); $devId=''
try{
  Say '== GlassWidgets enroll precheck (diagnostic only) =='
  if(-not (Test-Path $appExe)){ throw ('installed exe not found at '+$appExe) }
  $vi=(Get-Item $appExe).VersionInfo
  $verText=[string]$vi.ProductVersion; if([string]::IsNullOrWhiteSpace($verText)){ $verText=[string]$vi.FileVersion }
  Say ('app version: '+$verText)
  $m=[regex]::Match($verText,'^(\d+)\.(\d+)\.(\d+)')
  if(-not $m.Success){ $reasons+='cannot parse the installed version - evidence mode unavailable' }
  else{
    $verN=[int]$m.Groups[1].Value*1000000+[int]$m.Groups[2].Value*1000+[int]$m.Groups[3].Value
    if($verN -lt 4064){ $reasons+='installed '+$verText+' has no runtime-evidence mode (needs 0.4.64+) - the userData root cannot be proven' }
    else{
      # BUILD GATE (before ANY invocation): a version string does NOT prove the mode exists -
      # earlier 0.4.64 candidates lacked it and would boot normally if launched. The packed
      # resources\app.asar must (a) match the pinned hash when one is supplied, and (b) contain
      # the mode's build marker. An unknown build is NEVER launched.
      $asar=Join-Path (Join-Path $env:LOCALAPPDATA 'Programs\GlassWidgets') 'resources\app.asar'
      $buildOk=$false
      if(-not (Test-Path $asar)){ $reasons+='packed app.asar not found at '+$asar+' - unknown build, not launched' }
      else{
        if($AsarSha256){
          $h=(Get-FileHash -LiteralPath $asar -Algorithm SHA256).Hash
          if($h -ine $AsarSha256){ $reasons+='app.asar hash '+$h+' does NOT match the pinned build '+$AsarSha256+' - unknown build, not launched' }
          else{ Say ('build pinned: app.asar sha256 matches the reviewed build') ; $buildOk=$true }
        }else{ $reasons+='no -AsarSha256 pin supplied - build identity rests on the marker scan only' ; $buildOk=$true }
        if($buildOk){
          $marker='--gw-runtime-paths-json'
          $bytes=[System.IO.File]::ReadAllBytes($asar)
          $found=[System.Text.Encoding]::ASCII.GetString($bytes).IndexOf($marker) -ge 0
          if(-not $found){ $reasons+='build marker '+$marker+' ABSENT in the packed app.asar - this build has no evidence mode, NOT launched'; $buildOk=$false }
          else{ Say ('build marker present in app.asar: '+$marker) }
        }
      }
    if($buildOk){
      # 1. runtime evidence: the packed app answers for itself, nonce-bound
      $nonce=[guid]::NewGuid().ToString('N')
      $prevEap=$ErrorActionPreference; $ErrorActionPreference='Continue'
      try{ & cmd.exe /c ('"'+$appExe+'" --gw-runtime-paths-json '+$nonce+' >"'+$cap+'" 2>&1') | Out-Null }finally{ $ErrorActionPreference=$prevEap }
      $raw=''; if(Test-Path $cap){ $raw=(Get-Content $cap -Raw).Trim() }
      $jline=($raw -split "`r?`n" | Where-Object {$_.TrimStart().StartsWith('{')} | Select-Object -First 1)
      if(-not $jline){ $reasons+='the app did not return a runtime-paths JSON line (mode missing or blocked)' }
      else{
        $rt=$null; try{ $rt=$jline | ConvertFrom-Json -EA Stop }catch{ $reasons+='runtime-paths JSON did not parse' }
        if($rt){
          if($rt.v -ne 1){ $reasons+='runtime-paths schema is not v1' }
          elseif([string]$rt.nonce -ne $nonce){ $reasons+='nonce mismatch - the capture is not from this run (stale or foreign)' }
          elseif(-not ($rt.isPackaged -is [bool] -and $rt.isPackaged -eq $true)){ $reasons+='isPackaged is not strictly boolean true - this is not the packed installed runtime' }
          elseif([string]$rt.exe -ine $appExe){ $reasons+='the answering exe ('+[string]$rt.exe+') is not the installed exe ('+$appExe+')' }
          elseif(-not [regex]::Match([string]$rt.version,'^(\d+)\.(\d+)\.(\d+)').Success){ $reasons+='runtime version is unparseable: '+[string]$rt.version }
          elseif([regex]::Match([string]$rt.version,'^(\d+)\.(\d+)\.(\d+)').Value -cne $m.Value){ $reasons+='runtime version ('+[string]$rt.version+') does not match the installed version ('+$verText+')' }
          elseif(-not ($rt.ts -is [long] -or $rt.ts -is [int] -or $rt.ts -is [double])){ $reasons+='runtime ts is not numeric' }
          elseif([math]::Abs(([DateTimeOffset]::Now.ToUnixTimeMilliseconds())-[int64]$rt.ts) -gt 60000){ $reasons+='runtime ts is not fresh (over 60s from now) - stale capture rejected' }
          elseif([string]::IsNullOrWhiteSpace([string]$rt.userData) -or -not [System.IO.Path]::IsPathRooted([string]$rt.userData)){ $reasons+='runtime returned a missing or non-absolute userData' }
          else{
            $verdict='RUNTIME_EVIDENCE_OK'
            Say ('runtime userData (answered by the app itself): '+[string]$rt.userData)
            $devDir=Join-Path ([string]$rt.userData) 'device'
            $devFile=Join-Path $devDir 'device.json'
            # 2a. strict presence: only "not found" is absence; any other failure is unreadable
            $present=$false; $unreadable=$false
            try{ $null=Get-Item -LiteralPath $devFile -Force -ErrorAction Stop; $present=$true }
            catch [System.Management.Automation.ItemNotFoundException]{ $present=$false }
            catch [System.IO.DirectoryNotFoundException]{ $present=$false }
            catch [System.IO.FileNotFoundException]{ $present=$false }
            catch{ $unreadable=$true; $reasons+='cannot inspect device.json: '+$_.Exception.GetType().Name }
            if($unreadable){ $finding='unreadable' }
            elseif(-not $present){ $finding='absent'; if(Test-Path $devDir){ $reasons+='device directory exists but device.json does not' } }
            else{
              # 2b. ACL parity with src/lib/acl-check.js (evalAclDump): protected DACL, OWNER SID ==
              # current user, EXACTLY ONE ACE (sid == current user, Allow, IsInherited === false,
              # rights an integer in [0, 2032127]). Numeric, never string tests.
              $mySid=[System.Security.Principal.WindowsIdentity]::GetCurrent().User.Value
              function Test-OwnerOnly($p,$isDir){
                try{
                  $item=Get-Item -LiteralPath $p -Force
                  if(($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0){ return 'path is a reparse point/symlink: '+$p }
                  $a=Get-Acl -LiteralPath $p
                  if($a.AreAccessRulesProtected -ne $true){ return 'DACL not protected (inheritance on): '+$p }
                  $own=$null; try{ $own=([System.Security.Principal.NTAccount]$a.Owner).Translate([System.Security.Principal.SecurityIdentifier]).Value }catch{ $own=[string]$a.Owner }
                  if($own -ne $mySid){ return 'owner SID is not the current user on '+$p+': '+$own }
                  $rules=@($a.Access)
                  if($rules.Count -ne 1){ return 'ACE count '+$rules.Count+' on '+$p+' (expected exactly 1)' }
                  $r=$rules[0]
                  $sid=$null; try{ $sid=$r.IdentityReference.Translate([System.Security.Principal.SecurityIdentifier]).Value }catch{ $sid=[string]$r.IdentityReference }
                  if($sid -ne $mySid){ return 'ACE SID is not the current user on '+$p+': '+$sid }
                  if($r.AccessControlType -ne 'Allow'){ return 'ACE is not Allow on '+$p }
                  if([bool]$r.IsInherited -ne $false){ return 'ACE is inherited on '+$p }
                  $rights=[int64]$r.FileSystemRights
                  if($rights -lt 0 -or $rights -gt 2032127){ return 'rights out of bounds on '+$p+': '+$rights }
                  return ''
                }catch{ return 'ACL evaluation failed on '+$p+': '+$_.Exception.Message }
              }
              $aclF=Test-OwnerOnly $devFile $false
              $aclD=Test-OwnerOnly $devDir $true
              if($aclF){ $reasons+=('file ACL: '+$aclF) }
              if($aclD){ $reasons+=('dir ACL: '+$aclD) }
              # 2c. content health via the app's OWN runtime (same node as the app, no downloads)
              $contentOk=$false
              $js=@'
const fs=require('fs'),crypto=require('crypto');
try{
  const j=JSON.parse(fs.readFileSync(process.argv[2],'utf8'));
  const priv=crypto.createPrivateKey({key:Buffer.from(String(j.priv),'base64'),format:'der',type:'pkcs8'});
  const pub=crypto.createPublicKey(priv).export({format:'der',type:'spki'}).toString('base64');
  if(pub!==String(j.pub)){console.log('MISMATCH');}
  else{console.log('OK '+crypto.createHash('sha256').update(pub).digest('hex').slice(0,16));}
}catch(e){console.log('CORRUPT');}
'@
              Set-Content -Path $probe -Value $js -Encoding ASCII
              $prevEap2=$ErrorActionPreference; $ErrorActionPreference='Continue'
              try{ & cmd.exe /c ('set ELECTRON_RUN_AS_NODE=1&& "'+$appExe+'" "'+$probe+'" "'+$devFile+'" >"'+$probeOut+'" 2>&1') | Out-Null }finally{ $ErrorActionPreference=$prevEap2 }
              $res=''; if(Test-Path $probeOut){ $res=(Get-Content $probeOut -Raw).Trim() }
              if($res -match '^OK ([0-9a-f]{16})$'){ $contentOk=$true; $devId=$Matches[1] }
              elseif($res -eq 'MISMATCH'){ $reasons+='stored public key does NOT derive from the private key' }
              elseif($res -eq 'CORRUPT'){ $reasons+='device.json is not valid (unparseable / missing fields / bad key)' }
              else{ $reasons+='could not run the content probe (got: '+$res+')' }
              if($contentOk -and -not $aclF -and -not $aclD){ $finding='healthy' }
              else{ $finding='unhealthy' }
            }
          }
        }
      }
    }
    }
  }
  Say ''
  Say ('VERDICT: '+$verdict)
  Say ('FILE_FINDING: '+$finding)
  if($devId){ Say ('deviceId (derived): '+$devId) }
  foreach($r in $reasons){ Say ('  - '+$r) }
  Say ''
  Say 'Diagnostic only - no verdict or finding here is an enroll decision. Send this output back (and enroll-precheck-log.txt from the Desktop).'
}catch{
  Say ('FAIL: '+$_.Exception.Message)
  Say 'Send this whole output back.'
}finally{
  if($stage -and (Test-Path -LiteralPath $stage)){ Remove-Item -LiteralPath $stage -Recurse -Force -ErrorAction SilentlyContinue }
  try{ Stop-Transcript | Out-Null }catch{}
  Read-Host 'Press Enter to close'
}
