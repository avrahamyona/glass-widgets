# win-enroll-precheck.txt - run on the user's PC (regular Windows PowerShell 5.1, no admin)
# DIAGNOSTIC ONLY. Gathers evidence and findings; draws NO enroll conclusions, issues NO
# permits, changes NO app identity or settings. Writes a Desktop transcript
# (enroll-precheck-log.txt) and a unique GUID temp staging dir. The staging dir is removed
# at the end ONLY when containment closure is proven for everything that ran; otherwise it
# is PRESERVED and reported. Electron's own cache/log writes may happen - zero writes is
# NOT claimed. The bounded launcher is compiled LOCALLY by Add-Type (no downloads); if
# compilation is impossible the script stops at UNKNOWN before any launch.
#
# CONTAINMENT (the only kill scope): every launch runs inside a dedicated Windows Job
# Object with KILL_ON_JOB_CLOSE. The process is created SUSPENDED, assigned to the job,
# and only then resumed, so every descendant is contained from birth - no parent-pid
# enumeration, no late-descendant race, no reparent escape, no PID-reuse risk (the kernel
# tracks job membership by handle, not pid). Termination kills the whole job atomically
# and job emptiness is verified afterwards. A pre-existing running app instance is in a
# DIFFERENT job and can NEVER be touched by anything here. Output goes to files, never
# pipes, and files are read only after the run returns - reads are bounded by construction.
#
# PRE-LAUNCH GATE (ALL must pass before ANY invocation; otherwise UNKNOWN, nothing runs):
#   - -AsarSha256, -ExeSha256 AND -ExpectedVersion are ALL REQUIRED, injected at handoff
#     from the REVIEWED build only - never derived from source. An asar pin alone cannot
#     identify a REPLACED exe, so the packaged exe hash is pinned too.
#   - Installed exe version equals the reviewed version after PE normalization (a 4-part
#     PE FileVersion like 0.4.64.0 normalizes to 0.4.64; prerelease is kept).
#   - GlassWidgets.exe sha256 == -ExeSha256 AND resources\app.asar sha256 == -AsarSha256.
#   - The mode's build marker must be present in the pinned asar (supporting check only).
#
# LAUNCH CONTRACT (evidence mode AND content probe share it): suspended create, job
# assign, resume; per-child environment (ELECTRON_RUN_AS_NODE removed for the evidence
# launch, set to 1 ONLY for the probe child); bounded wait (evidence 20s, probe 10s);
# natural exit must be code 0 with ZERO active processes left in the job (a surviving
# descendant means the capture may be contaminated - rejected); the whole trimmed stdout
# is parsed ONCE as exactly one JSON document (arrays, scalars, multiple documents,
# leading/trailing nonblank content and malformed JSON are all rejected).
# ACL comparison mirrors the app's shared evaluator (src/lib/acl-check.js) exactly.
param([string]$AsarSha256='',[string]$ExeSha256='',[string]$ExpectedVersion='')

using System.Text;
using System.Runtime.InteropServices;

public static class GwBounded {
  public sealed class Result {
    public bool Started;
    public bool TimedOut;
    public bool ClosureProven;
    public int ActiveAfterExit;
    public int ExitCode;
    public string Error;
  }

  const uint CREATE_SUSPENDED = 0x00000004;
  const uint CREATE_NO_WINDOW = 0x08000000;
  const uint CREATE_UNICODE_ENVIRONMENT = 0x00000400;
  const uint STARTF_USESTDHANDLES = 0x00000100;
  const uint WAIT_TIMEOUT = 0x00000102;
  const uint GENERIC_READ = 0x80000000;

  const int JobObjectExtendedLimitInformation = 9;
  const int JobObjectBasicAccountingInformation = 1;

  [StructLayout(LayoutKind.Sequential)]
  struct SECURITY_ATTRIBUTES { public int nLength; public IntPtr lpSecurityDescriptor; public bool bInheritHandle; }

  [StructLayout(LayoutKind.Sequential, CharSet=CharSet.Unicode)]

    public string lpReserved; public string lpDesktop; public string lpTitle;

    public short wShowWindow; public short cbReserved2; public IntPtr lpReserved2;
    public IntPtr hStdInput; public IntPtr hStdOutput; public IntPtr hStdError;
  }

  [StructLayout(LayoutKind.Sequential)]

  public static string Quote(string exe, string[] args) {
    StringBuilder sb = new StringBuilder();
    AppendArg(sb, exe);
    if (args != null) { foreach (string a in args) { sb.Append(' '); AppendArg(sb, a); } }
    return sb.ToString();

  static void AppendArg(StringBuilder sb, string a) {
    if (a == null) { a = ""; }
    if (a.Length != 0 && a.IndexOfAny(new char[] { ' ', '\t', '\n', '\v', '"' }) < 0) { sb.Append(a); return; }

      if (a[i] == '"') { for (int k = 0; k < bs * 2 + 1; k++) { sb.Append('\\'); } sb.Append('"'); i++; }

      if (job == IntPtr.Zero) { r.Error = Err("CreateJobObject"); return r; }

      li.BasicLimitInformation.LimitFlags = JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE;
      int liSize = Marshal.SizeOf(typeof(JOBOBJECT_EXTENDED_LIMIT_INFORMATION));

      si.cb = Marshal.SizeOf(typeof(STARTUPINFO));

      r.Started = true;

        TerminateProcess(proc, 1);
        WaitForSingleObject(proc, 5000);
        r.ClosureProven = (WaitForSingleObject(proc, 0) == 0);
        r.Error = Err("AssignProcessToJobObject");
        return r;
      }
      ResumeThread(thread);
      CloseHandle(thread); thread = IntPtr.Zero;
      uint w = WaitForSingleObject(proc, (uint)timeoutMs);
      if (w == WAIT_TIMEOUT) {
        r.TimedOut = true;
        TerminateJobObject(job, 1);
        WaitForSingleObject(proc, 5000);
      } else if (w != 0) {
        r.Error = Err("WaitForSingleObject");
        TerminateJobObject(job, 1);
        WaitForSingleObject(proc, 5000);
      }
      int active = QueryActive(job);
      r.ActiveAfterExit = active < 0 ? -1 : active;
      if (active != 0) {
        TerminateJobObject(job, 1);
        for (int i = 0; i < 50; i++) {
          active = QueryActive(job);
          if (active == 0) { break; }
          if (active < 0) { break; }
          System.Threading.Thread.Sleep(100);
        }
      }
      r.ClosureProven = (active == 0);
      if (!r.TimedOut && r.Error.Length == 0) {
        uint code;
        if (GetExitCodeProcess(proc, out code)) { r.ExitCode = (int)code; }
      }
      return r;
    } finally {
      if (thread != IntPtr.Zero) { CloseHandle(thread); }
      if (proc != IntPtr.Zero) { CloseHandle(proc); }
      if (job != IntPtr.Zero) { CloseHandle(job); }
      if (hIn != IntPtr.Zero && hIn != INVALID_HANDLE) { CloseHandle(hIn); }
      if (hOut != IntPtr.Zero && hOut != INVALID_HANDLE) { CloseHandle(hOut); }
      if (hErr != IntPtr.Zero && hErr != INVALID_HANDLE) { CloseHandle(hErr); }
    }
  }
}

# ---------- pure, testable helpers (no side effects; delegates injectable) ----------
function ConvertTo-NormalizedVersion {
  param([AllowEmptyString()][string]$v)
  $t=([string]$v).Trim()
  $core=$t; $pre=''
  $mi=[regex]::Match($t,'^([^-]+)(-.*)$')
  if($mi.Success){ $core=$mi.Groups[1].Value; $pre=$mi.Groups[2].Value }
  $parts=@($core.Split('.'))
  while($parts.Count -gt 3 -and $parts[$parts.Count-1] -eq '0'){ $parts=$parts[0..($parts.Count-2)] }
  return (($parts -join '.')+$pre)
}
function Test-VersionGate {
  param([AllowEmptyString()][string]$VerText,[AllowEmptyString()][string]$ExpectedVersion)
  if([string]::IsNullOrWhiteSpace($VerText)){ return 'cannot read the installed exe version - evidence unavailable' }
  $a=ConvertTo-NormalizedVersion $VerText
  $b=ConvertTo-NormalizedVersion $ExpectedVersion
  if($a -cne $b){ return ('installed exe version ('+$VerText+', normalized '+$a+') is not exactly the reviewed version ('+$ExpectedVersion+') - not launched') }
  return ''
}
function ConvertTo-RuntimeEvidence {
  param([AllowNull()][AllowEmptyString()][string]$Raw,[string]$Nonce,[string]$ExpectedVersion,[string]$AppExe,[long]$NowMs)
  $t=([string]$Raw).Trim()
  if($t -eq ''){ return [pscustomobject]@{ok=$false;reason='empty capture - no JSON record at all';rt=$null} }
  if($t.StartsWith('[')){ return [pscustomobject]@{ok=$false;reason='capture is a JSON array - exactly one object required';rt=$null} }
  $rt=$null
  try{ $rt=($t | ConvertFrom-Json -ErrorAction Stop) }catch{ return [pscustomobject]@{ok=$false;reason=('capture is not exactly one valid JSON document: '+$_.Exception.Message);rt=$null} }
  if($null -eq $rt){ return [pscustomobject]@{ok=$false;reason='capture parsed to null';rt=$null} }
  if($rt -is [array]){ return [pscustomobject]@{ok=$false;reason='capture is a JSON array - exactly one object required';rt=$null} }
  if(-not ($rt -is [pscustomobject])){ return [pscustomobject]@{ok=$false;reason='capture is not a JSON object (scalar rejected)';rt=$null} }
  if($rt.v -ne 1){ return [pscustomobject]@{ok=$false;reason='runtime-paths schema is not v1';rt=$null} }
  if([string]$rt.nonce -cne $Nonce){ return [pscustomobject]@{ok=$false;reason='nonce mismatch - the capture is not from this run (stale or foreign)';rt=$null} }
  if(-not ($rt.isPackaged -is [bool] -and $rt.isPackaged)){ return [pscustomobject]@{ok=$false;reason='isPackaged is not strictly boolean true - this is not the packed installed runtime';rt=$null} }
  if([string]$rt.exe -ine $AppExe){ return [pscustomobject]@{ok=$false;reason=('the answering exe ('+[string]$rt.exe+') is not the installed exe ('+$AppExe+')');rt=$null} }
  if([string]$rt.version -cne $ExpectedVersion.Trim()){ return [pscustomobject]@{ok=$false;reason=('runtime version ('+[string]$rt.version+') does not exactly match the reviewed package version ('+$ExpectedVersion.Trim()+')');rt=$null} }
  if(-not ($rt.ts -is [long] -or $rt.ts -is [int] -or $rt.ts -is [double])){ return [pscustomobject]@{ok=$false;reason='runtime ts is not numeric';rt=$null} }
  if([math]::Abs($NowMs-[int64]$rt.ts) -gt 60000){ return [pscustomobject]@{ok=$false;reason='runtime ts is not fresh (over 60s from now) - stale capture rejected';rt=$null} }
  $ud=[string]$rt.userData
  if([string]::IsNullOrWhiteSpace($ud) -or -not ($ud -match '^([A-Za-z]:[\\/]|\\\\)')){ return [pscustomobject]@{ok=$false;reason='runtime returned a missing or non-absolute userData';rt=$null} }
  return [pscustomobject]@{ok=$true;reason='';rt=$rt}
}
function ConvertTo-ProbeResult {
  param([AllowEmptyString()][string]$Raw)
  $t=([string]$Raw).Trim()
  if($t -cmatch '^OK ([0-9a-f]{16})$'){ return [pscustomobject]@{state='ok';deviceId=$Matches[1]} }
  if($t -eq 'MISMATCH'){ return [pscustomobject]@{state='mismatch';deviceId=''} }
  if($t -eq 'CORRUPT'){ return [pscustomobject]@{state='corrupt';deviceId=''} }
  return [pscustomobject]@{state='error';deviceId=''}
}
function Test-ClosureProven {
  # The staging dir may be removed only when EVERY started run ended with verified job
  # emptiness. A run that never started left nothing behind. Anything else: preserve.
  param($Runs)
  foreach($r in @($Runs)){ if($r.started -and -not $r.treeCleaned){ return $false } }
  return $true
}
function Get-RunAssessment {
  # '' when the run's output is usable, else the rejection reason. Every adversarial
  # outcome is rejected explicitly: launcher API failure, never started, timeout
  # (contained or unverifiable), nonzero exit, and a natural exit that left surviving
  # descendants in the job (they inherited the capture files - possible contamination).
  param($Run,[string]$Label)
  if([string]$Run.apiError){ return ('bounded launcher failed for '+$Label+': '+[string]$Run.apiError) }
  if(-not $Run.started){ return ($Label+' could not be started') }
  if($Run.timedOut){
    if($Run.treeCleaned){ return ($Label+' timed out - the whole job was terminated and verified empty; rejected') }
    return ($Label+' timed out and job emptiness could NOT be verified - rejected; staging preserved, check Task Manager')
  }
  if($Run.exitCode -ne 0){ return ($Label+' exited with code '+$Run.exitCode+' - rejected') }
  if($Run.activeAfterExit -ne 0){ return ($Label+' exited but left active descendants in its job (capture may be contaminated) - rejected; descendants were terminated inside the job') }
  return ''
}
function Invoke-WithChildEnv {
  # Temporarily applies per-child env overrides to THIS process (CreateProcess copies the
  # parent env at creation time), invokes Body, then restores every touched variable -
  # including on exceptions. The child sees exactly the intended env; the parent is restored.
  param([hashtable]$EnvRemove,[hashtable]$EnvSet,[scriptblock]$Body)
  $saved=@{}
  foreach($k in (@($EnvRemove.Keys)+@($EnvSet.Keys))){ $saved[$k]=[Environment]::GetEnvironmentVariable($k,'Process') }
  try{
    foreach($k in $EnvRemove.Keys){ Remove-Item -LiteralPath ('Env:'+$k) -ErrorAction SilentlyContinue }
    foreach($k in $EnvSet.Keys){ [Environment]::SetEnvironmentVariable($k,[string]$EnvSet[$k],'Process') }
    & $Body
  }finally{
    foreach($k in $saved.Keys){
      if($null -eq $saved[$k]){ Remove-Item -LiteralPath ('Env:'+$k) -ErrorAction SilentlyContinue }
      else{ [Environment]::SetEnvironmentVariable($k,[string]$saved[$k],'Process') }
    }
  }
}
function Invoke-BoundedProcess {
  # Shared bounded-launch contract: the child runs inside a dedicated kill-on-close Job
  # Object (created suspended, assigned, resumed - via GwBounded, compiled locally).
  # Output goes to FILES, never pipes, and is read only after the run returns, so reads
  # are bounded by construction. Runner is injectable for deterministic tests.
  param(
    [Parameter(Mandatory=$true)][string]$ExePath,
    [string[]]$Arguments=@(),
    [Parameter(Mandatory=$true)][string]$OutDir,
    [hashtable]$EnvRemove=@{},
    [hashtable]$EnvSet=@{},
    [int]$TimeoutMs=20000,
    [scriptblock]$Runner=$null
  )
  $outPath=Join-Path $OutDir ('stdout-'+[guid]::NewGuid().ToString('N')+'.txt')
  $errPath=Join-Path $OutDir ('stderr-'+[guid]::NewGuid().ToString('N')+'.txt')
  $res=$null
  if($null -ne $Runner){
    $res=& $Runner $ExePath $Arguments $outPath $errPath $TimeoutMs
  }else{
    try{
      $res=Invoke-WithChildEnv -EnvRemove $EnvRemove -EnvSet $EnvSet -Body {
        $r=[GwBounded]::Run($ExePath,[string[]]$Arguments,$outPath,$errPath,$TimeoutMs)
        [pscustomobject]@{started=[bool]$r.Started;timedOut=[bool]$r.TimedOut;treeCleaned=[bool]$r.ClosureProven;activeAfterExit=[int]$r.ActiveAfterExit;exitCode=[int]$r.ExitCode;apiError=[string]$r.Error}
      }.GetNewClosure()
    }catch{
      $res=[pscustomobject]@{started=$false;timedOut=$false;treeCleaned=$false;activeAfterExit=-1;exitCode=$null;apiError=('launcher exception: '+$_.Exception.Message)}
    }
  }
  $out=''; if(Test-Path -LiteralPath $outPath){ try{ $out=[System.IO.File]::ReadAllText($outPath) }catch{ $out='' } }
  $err=''; if(Test-Path -LiteralPath $errPath){ try{ $err=[System.IO.File]::ReadAllText($errPath) }catch{ $err='' } }
  return [pscustomobject]@{started=[bool]$res.started;timedOut=[bool]$res.timedOut;treeCleaned=[bool]$res.treeCleaned;activeAfterExit=[int]$res.activeAfterExit;exitCode=$res.exitCode;stdout=$out;stderr=$err;apiError=[string]$res.apiError}
}

      $ha=(Get-FileHash -LiteralPath $asar -Algorithm SHA256).Hash
      if($hx -ine $ExeSha256){ $reasons+='GlassWidgets.exe hash '+$hx+' does NOT match the pinned reviewed exe '+$ExeSha256+' - unknown build, not launched' }
      elseif($ha -ine $AsarSha256){ $reasons+='app.asar hash '+$ha+' does NOT match the pinned reviewed asar '+$AsarSha256+' - unknown build, not launched' }
      else{
        Say 'build pinned: exe AND app.asar sha256 match the reviewed build'
        $marker='--gw-runtime-paths-json'
        $bytes=[System.IO.File]::ReadAllBytes($asar)
        $found=[System.Text.Encoding]::ASCII.GetString($bytes).IndexOf($marker) -ge 0
        if(-not $found){ $reasons+='build marker '+$marker+' ABSENT in the pinned app.asar - NOT launched (inconsistent build)' }
        else{ Say ('build marker present (supporting check): '+$marker); $gateOk=$true }
      }
    }
  }
  $rt=$null
  if($gateOk){
    $nonce=[guid]::NewGuid().ToString('N')
    $ev=Invoke-BoundedProcess -ExePath $appExe -Arguments @('--gw-runtime-paths-json',$nonce) -OutDir $stage -EnvRemove @{ELECTRON_RUN_AS_NODE=1} -TimeoutMs 20000
    $runs+=$ev
    $ra=Get-RunAssessment -Run $ev -Label 'evidence mode (20s bound)'
    if($ra){ $reasons+=$ra }
    else{
      $nowMs=[DateTimeOffset]::Now.ToUnixTimeMilliseconds()
      $pr=ConvertTo-RuntimeEvidence -Raw $ev.stdout -Nonce $nonce -ExpectedVersion $ExpectedVersion -AppExe $appExe -NowMs $nowMs
      if(-not $pr.ok){ $reasons+=$pr.reason }
      else{
        $rt=$pr.rt
        $verdict='RUNTIME_EVIDENCE_OK'
        Say ('runtime userData (answered by the app itself): '+[string]$rt.userData)
      }
    }
  }
  if($rt){
    $devDir=Join-Path ([string]$rt.userData) 'device'
    $devFile=Join-Path $devDir 'device.json'
    $present=$false; $unreadable=$false
    try{ $null=Get-Item -LiteralPath $devFile -Force -ErrorAction Stop; $present=$true }
    catch [System.Management.Automation.ItemNotFoundException]{ $present=$false }
    catch [System.IO.DirectoryNotFoundException]{ $present=$false }
    catch [System.IO.FileNotFoundException]{ $present=$false }
    catch{ $unreadable=$true; $reasons+='cannot inspect device.json: '+$_.Exception.GetType().Name }
    if($unreadable){ $finding='unreadable' }
    elseif(-not $present){ $finding='absent'; if(Test-Path $devDir){ $reasons+='device directory exists but device.json does not' } }
    else{
      # ACL parity with src/lib/acl-check.js (evalAclDump): protected DACL, OWNER SID ==
      # current user, EXACTLY ONE ACE (sid == current user, Allow, IsInherited === false,
      # rights an integer in [0, 2032127]). Numeric, never string tests.
      $mySid=[System.Security.Principal.WindowsIdentity]::GetCurrent().User.Value
      function Test-OwnerOnly($p,$isDir){
        try{
          $item=Get-Item -LiteralPath $p -Force
          if(($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0){ return 'path is a reparse point/symlink: '+$p }
          $a=Get-Acl -LiteralPath $p
          if($a.AreAccessRulesProtected -ne $true){ return 'DACL not protected (inheritance on): '+$p }
          $own=$null; try{ $own=([System.Security.Principal.NTAccount]$a.Owner).Translate([System.Security.Principal.SecurityIdentifier]).Value }catch{ $own=[string]$a.Owner }
          if($own -ne $mySid){ return 'owner SID is not the current user on '+$p+': '+$own }
          $rules=@($a.Access)
          if($rules.Count -ne 1){ return 'ACE count '+$rules.Count+' on '+$p+' (expected exactly 1)' }
          $r=$rules[0]
          $sid=$null; try{ $sid=$r.IdentityReference.Translate([System.Security.Principal.SecurityIdentifier]).Value }catch{ $sid=[string]$r.IdentityReference }
          if($sid -ne $mySid){ return 'ACE SID is not the current user on '+$p+': '+$sid }
          if($r.AccessControlType -ne 'Allow'){ return 'ACE is not Allow on '+$p }
          if([bool]$r.IsInherited -ne $false){ return 'ACE is inherited on '+$p }
          $rights=[int64]$r.FileSystemRights
          if($rights -lt 0 -or $rights -gt 2032127){ return 'rights out of bounds on '+$p+': '+$rights }
          return ''
        }catch{ return 'ACL evaluation failed on '+$p+': '+$_.Exception.Message }
      }
      $aclF=Test-OwnerOnly $devFile $false
      $aclD=Test-OwnerOnly $devDir $true
      if($aclF){ $reasons+=('file ACL: '+$aclF) }
      if($aclD){ $reasons+=('dir ACL: '+$aclD) }
      # content health via the app's OWN runtime - SAME containment contract: job-scoped
      # launch, ELECTRON_RUN_AS_NODE=1 for THIS child only, 10s bound, exit 0, zero active
      # processes left in the job, and an exact known output line.
      $contentOk=$false
      $js=@'
const fs=require('fs'),crypto=require('crypto');
try{
  const j=JSON.parse(fs.readFileSync(process.argv[2],'utf8'));
  const priv=crypto.createPrivateKey({key:Buffer.from(String(j.priv),'base64'),format:'der',type:'pkcs8'});
  const pub=crypto.createPublicKey(priv).export({format:'der',type:'spki'}).toString('base64');
  if(pub!==String(j.pub)){console.log('MISMATCH');}
  else{console.log('OK '+crypto.createHash('sha256').update(pub).digest('hex').slice(0,16));}
}catch(e){console.log('CORRUPT');}
'@
      Set-Content -Path $probe -Value $js -Encoding ASCII
      $pv=Invoke-BoundedProcess -ExePath $appExe -Arguments @($probe,$devFile) -OutDir $stage -EnvSet @{ELECTRON_RUN_AS_NODE='1'} -TimeoutMs 10000
      $runs+=$pv
      $ra2=Get-RunAssessment -Run $pv -Label 'content probe (10s bound)'
      if($ra2){ $reasons+=$ra2 }
      else{
        $cpr=ConvertTo-ProbeResult -Raw $pv.stdout
        if($cpr.state -eq 'ok'){ $contentOk=$true; $devId=$cpr.deviceId }
        elseif($cpr.state -eq 'mismatch'){ $reasons+='stored public key does NOT derive from the private key' }
        elseif($cpr.state -eq 'corrupt'){ $reasons+='device.json is not valid (unparseable / missing fields / bad key)' }
        else{ $reasons+='content probe returned an unrecognized output (not the exact known line) - rejected' }
      }
      if($contentOk -and -not $aclF -and -not $aclD){ $finding='healthy' }
      else{ $finding='unhealthy' }
    }
  }
  Say ''
  Say ('VERDICT: '+$verdict)
  Say ('FILE_FINDING: '+$finding)
  if($devId){ Say ('deviceId (derived): '+$devId) }
  foreach($r in $reasons){ Say ('  - '+$r) }
  Say ''
  Say 'Diagnostic only - no verdict or finding here is an enroll decision. Send this output back (and enroll-precheck-log.txt from the Desktop).'
}catch{
  Say ('FAIL: '+$_.Exception.Message)
  Say 'Send this whole output back.'
}finally{
  if($stage -and (Test-Path -LiteralPath $stage)){
    if(Test-ClosureProven $runs){
      Remove-Item -LiteralPath $stage -Recurse -Force -ErrorAction SilentlyContinue
    }else{
      Say ('STAGING PRESERVED at '+$stage+' - process containment closure was NOT proven; do not delete manually before checking Task Manager for leftover processes, and report this line back.')
    }
  }
  try{ Stop-Transcript | Out-Null }catch{}
  Read-Host 'Press Enter to close'
}
