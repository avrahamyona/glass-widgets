# win-watchdog-remove.txt - run on the user's PC (regular PowerShell, no admin needed)
# Purpose: REMOVE the GlassWidgets-Watchdog scheduled task (the every-5-minutes relauncher).
# Per the user's instruction: the app must NOT reopen itself every 5 minutes; opening stays in
# his hands. This script DELETES the task and VERIFIES it is gone. It does NOT launch the app,
# does NOT re-create the task, and does NOT touch autostart (the Run key) or the app itself.
# Verification is exact, not exit-code-based: the Task Scheduler API (Schedule.Service) is asked
# for the task, and ONLY the precise "not found" (0x80070002) counts as absent. Access denied,
# a scheduler outage, or any other failure is NOT treated as absence - the script fails closed.
# Idempotent: a verified-absent task is a PASS. Undo (re-create) is printed at the end.
$ErrorActionPreference='Stop'
function Say($m){ Write-Host $m }
$taskName='GlassWidgets-Watchdog'
$HR_NOT_FOUND=-2147024894 # 0x80070002 - "the system cannot find the file specified" (task truly absent)
$script:svc=$null
function WdFolder(){
  if(-not $script:svc){ $script:svc=New-Object -ComObject 'Schedule.Service'; $script:svc.Connect() }
  return $script:svc.GetFolder('\')
}
function Test-WdTaskPresent(){
  # returns $true/$false ONLY on a definitive answer; anything else throws (fail closed)
  try{ $null=(WdFolder).GetTask($taskName); return $true }
  catch{
    $hr=$_.Exception.HResult
    if($hr -eq $HR_NOT_FOUND -or ([string]$_.Exception.Message) -match '0x80070002'){ return $false }
    throw ('task query failed - this is NOT proof of absence ('+$_.Exception.Message+')')
  }
}
try{
  Say '== GlassWidgets watchdog removal =='
  # PASS 1: delete the task if (and only if) it provably exists
  if(-not (Test-WdTaskPresent)){
    Say 'PASS 1: GlassWidgets-Watchdog is provably absent (exact not-found) - nothing to delete'
  }else{
    $deleted=$false
    try{ (WdFolder).DeleteTask($taskName,0); $deleted=$true }
    catch{
      $hr=$_.Exception.HResult
      if($hr -eq $HR_NOT_FOUND -or ([string]$_.Exception.Message) -match '0x80070002'){
        Say 'PASS 1: task vanished before the delete call (exact not-found) - goal reached'
      }else{ throw ('task delete failed ('+$_.Exception.Message+') - NOT deleted, NOT assumed gone') }
    }
    if($deleted){ Say 'PASS 1: delete call issued for GlassWidgets-Watchdog' }
  }
  # PASS 2: verify AFTER the delete with a FRESH query - only the exact not-found proves it
  if(Test-WdTaskPresent){ throw 'task still present after the delete call - removal failed, nothing assumed' }
  Say 'PASS 2: verified - the task no longer exists (exact not-found on a fresh query)'
  Say '      nothing will relaunch the app every 5 minutes anymore.'
  Say ''
  Say 'NOT touched (on purpose): the app itself, its data, and the login autostart (Run key).'
  Say 'Opening the app is now fully in your hands - nothing relaunches it on a timer.'
  Say ''
  Say '--- undo (only if you ever want the auto-relaunch back) ---'
  Say '  re-run win-server-base, which re-creates the script and the task.'
  Say ''
  Say 'DONE: watchdog removed. Send this output back.'
}catch{
  Say ('FAIL: '+$_.Exception.Message)
  Say 'Nothing was assumed - if the task may still exist, this says so. Send this whole output back.'
}finally{
  Read-Host 'Press Enter to close'
}
