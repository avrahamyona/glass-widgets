# GlassWidgets listen probe (temporary, reversible, no product code)
# What: listens on a TCP port (default 47700) on ALL local addresses (IPv6 dual-stack, which
# also serves IPv4 and localhost), logs every inbound connection with its remote endpoint,
# and answers a minimal HTTP 200 so a phone browser or an external port checker can see it.
# With -OpenFirewall it also adds ONE inbound Windows-firewall rule for this port and
# REMOVES that rule again when the listener stops. Nothing else is changed; no files are
# written except the log next to this script. Authorization is established outside this file.
# Run in PowerShell (as Administrator when using -OpenFirewall), then stop with Ctrl+C
# or let the -Minutes window expire.
param([int]$Port=47700,[int]$Minutes=10,[switch]$OpenFirewall)
$ErrorActionPreference='Continue'
$ruleName='GW-NetListen-Probe'
$ruleMade=$false
$logFile=Join-Path $PSScriptRoot 'gw-netlisten-log.txt'
function Log([string]$m){ $line=(Get-Date -Format 'HH:mm:ss')+' '+$m; Write-Host $line; Add-Content -Path $logFile -Value $line -Encoding UTF8 }
Log(('== GlassWidgets listen probe on TCP port {0} for {1} min ==' -f $Port,$Minutes))
Log(('PS version: '+$PSVersionTable.PSVersion))
# show this PC's candidate addresses so the tester knows what to browse to
try { Get-NetIPAddress -AddressFamily IPv6 -ErrorAction Stop | Where-Object { $_.PrefixOrigin -eq 'RouterAdvertisement' -or ($_.IPAddress -like '2*') -or ($_.IPAddress -like '3*') } | ForEach-Object { Log(('global IPv6: '+$_.IPAddress+' ('+$_.InterfaceAlias+')')) } } catch { Log('UNVERIFIED: IPv6 list failed (not Windows?): '+$_.Exception.Message) }
try { Get-NetIPAddress -AddressFamily IPv4 -ErrorAction Stop | Where-Object { $_.IPAddress -notlike '127.*' -and $_.IPAddress -notlike '169.254.*' } | ForEach-Object { Log(('local IPv4: '+$_.IPAddress+' ('+$_.InterfaceAlias+')')) } } catch { Log('UNVERIFIED: IPv4 list failed (not Windows?): '+$_.Exception.Message) }
if($OpenFirewall){
  $isAdmin=([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
  if(-not $isAdmin){ Log('ERROR: -OpenFirewall needs an elevated PowerShell (Run as administrator). No rule created.'); exit 2 }
  try {
    New-NetFirewallRule -DisplayName $ruleName -Direction Inbound -Protocol TCP -LocalPort $Port -Action Allow -Profile Any -ErrorAction Stop | Out-Null
    $ruleMade=$true; Log(('firewall: inbound TCP rule '+$ruleName+' on port '+$Port+' CREATED (will be removed on exit)'))
  } catch { Log(('ERROR: firewall rule creation failed: '+$_.Exception.Message)); exit 2 }
} else { Log('firewall: untouched (run with -OpenFirewall to add a temporary inbound rule)') }
$listener=New-Object System.Net.Sockets.TcpListener([System.Net.IPAddress]::IPv6Any,$Port)
try { $listener.Server.DualMode=$true } catch { Log('note: DualMode not settable here; IPv6 only') }
try { $listener.Start() } catch {
  Log(('ERROR: cannot bind port '+$Port+': '+$_.Exception.Message))
  Log('hint: if the GlassWidgets app or standalone server is already running, it may hold this port. Close it, or rerun with -Port 47701.')
  if($ruleMade){ try { Remove-NetFirewallRule -DisplayName $ruleName -ErrorAction Stop; Log('firewall: rule removed after bind failure') } catch { Log(('WARN: rule removal failed: '+$_.Exception.Message)) } }
  exit 3
}
Log(('listening on [::]:'+$Port+' (dual-stack: IPv6 + IPv4 + localhost). Waiting for connections...'))
Log(('test from the phone (mobile data, Wi-Fi OFF): http://[GLOBAL-IPV6]:'+$Port+'/  and  http://EXTERNAL-IPV4:'+$Port+'/'))
$deadline=(Get-Date).AddMinutes($Minutes)
$hits=0
try {
  while((Get-Date) -lt $deadline){
    if($listener.Pending()){
      $client=$listener.AcceptTcpClient()
      $remote='unknown'
      try { $remote=$client.Client.RemoteEndPoint.ToString() } catch {}
      $reqline='(no data)'
      try {
        $stream=$client.GetStream(); $stream.ReadTimeout=1500
        $buf=New-Object byte[] 4096
        if($stream.DataAvailable -or $client.Available -gt 0){
          $n=$stream.Read($buf,0,$buf.Length)
          if($n -gt 0){ $reqline=([Text.Encoding]::ASCII.GetString($buf,0,$n) -split "[`r`n]")[0] }
        }
        $body='{"ok":true,"probe":"gw-netlisten","port":'+$Port+'}'
        $resp="HTTP/1.1 200 OK`r`nContent-Type: application/json`r`nContent-Length: $($body.Length)`r`nConnection: close`r`n`r`n$body"
        $bytes=[Text.Encoding]::ASCII.GetBytes($resp)
        $stream.Write($bytes,0,$bytes.Length)
      } catch {}
      $hits++
      Log(('HIT #'+$hits+' from '+$remote+'  '+$reqline))
      try { $client.Close() } catch {}
    } else { Start-Sleep -Milliseconds 200 }
  }
} finally {
  try { $listener.Stop() } catch {}
  if($ruleMade){ try { Remove-NetFirewallRule -DisplayName $ruleName -ErrorAction Stop; Log(('firewall: rule '+$ruleName+' REMOVED')) } catch { Log(('WARN: rule removal failed - remove manually: Remove-NetFirewallRule -DisplayName '+$ruleName+' | error: '+$_.Exception.Message)) } }
}
Log(('== done: '+$hits+' inbound connection(s) in '+$Minutes+' min. Log saved to '+$logFile+' =='))
