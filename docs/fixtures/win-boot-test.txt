# בדיקת Boot עצמאית - "החיים שלי השרתי" (שלד השרת על Windows) - גרסה 10 (SHA נעוץ b78d70e ללא שינוי, SID מדויק+dir, אימות חתימת health, שוויון keypair, ייחוס PASS7; תיקון v7: דרישת ה-rights לקובץ הותאמה לחוזה הקוד - R+W בכוונת least-privilege, FullControl רק לתיקייה; תיקון v8: verify-sig.js עם String.fromCharCode(10); תיקון v9: PASS 7 מייחס משני הזרמים; תיקון v10: החוזה האמיתי = exit לא-אפס + אין listener + הזהות לא שוכתבה (SyntaxError גולמי של b78d70e מקובל) + ProgressPreference מושתק)
# =================================================================
#
# הרצה (דרך אחת מדויקת):
#   1. שמור את הקובץ הזה בשם boot-test.ps1 על שולחן העבודה
#   2. פתח PowerShell (תפריט התחל -> PowerShell)
#   3. הדבק שורה אחת:
#      powershell -ExecutionPolicy Bypass -File "$env:USERPROFILE\Desktop\boot-test.ps1"
#   4. החלון לא ייסגר לבד - בסוף תתבקש ללחוץ Enter
#   5. שלח בחזרה: את כל הטקסט על המסך, וגם את הקובץ boot-test-log.txt
#      שנוצר לצד הסקריפט (על שולחן העבודה)
#
# פלט צפוי - 7 שורות PASS ו-DONE:
#   PASS 1: node נמצא (v...)
#   PASS 2: הקוד ירד ונפתח
#   PASS 3: boot ראשון עלה + זהות נוצרה (device <16 תווים hex>)
#   PASS 4: ACL על קובץ הזהות - משתמש נוכחי בלבד
#   PASS 5: /health בלי מפתח -> 403, עם מפתח -> 200
#   PASS 6: boot שני - אותה זהות בדיוק
#   PASS 7: זהות משובשת -> סירוב עלייה (fail-loud)
#   DONE - כל הבדיקות עברו
#
# גם אם משהו נכשל - הכל נשמר ב-boot-test-log.txt. שלח אותו כמו שהוא.
# =================================================================
$ErrorActionPreference='Stop'`n$ProgressPreference='SilentlyContinue' # no progress CLIXML in redirected child streams
$scriptDir=$PSScriptRoot; if(-not $scriptDir){$scriptDir=(Get-Location).Path}
$logPath=Join-Path $scriptDir 'boot-test-log.txt'
try{Start-Transcript -Path $logPath -Force | Out-Null}catch{}
[Net.ServicePointManager]::SecurityProtocol=[Net.SecurityProtocolType]::Tls12
function Say($m){Write-Host $m}
$script:gwPids=@()
$base=Join-Path $env:TEMP ('gw-boot-'+[guid]::NewGuid().ToString('N').Substring(0,8))
New-Item -ItemType Directory -Path $base -Force | Out-Null
$code=Join-Path $base 'code'; $data=Join-Path $base 'data'
try{
  $nv=(node -v 2>$null); if(-not $nv){throw 'node not found in PATH'}
  Say ("PASS 1: node found ({0})" -f $nv)
  $zip=Join-Path $base 'gw.zip'
  $pinnedSha='b78d70e67271be498b712103fffd5cda3730faae' # the EXACT reviewed snapshot - never a moving branch
  Invoke-WebRequest -Uri ('https://codeload.github.com/avrahamyona/glass-widgets/zip/'+$pinnedSha) -OutFile $zip -UseBasicParsing
  Expand-Archive -Path $zip -DestinationPath $code -Force
  $root=Get-ChildItem $code -Directory | Select-Object -First 1
  Say ('PASS 2: code downloaded and extracted (pinned '+$pinnedSha.Substring(0,7)+')')
  $pubFile=Join-Path $base 'sender-pub.txt'
  node -e "const c=require('crypto');const{publicKey}=c.generateKeyPairSync('ed25519');require('fs').writeFileSync(process.argv[1],publicKey.export({format:'der',type:'spki'}).toString('base64'))" $pubFile
  if($LASTEXITCODE -ne 0){throw 'sender keypair generation failed'}
  $senderPub=(Get-Content $pubFile -Raw).Trim()
  $env:GW_SERVER_TOKEN='boot-test-token-123'
  $env:GW_SENDER_PUB=$senderPub
  $env:GW_SERVER_DATA=$data
  $env:GW_SERVER_PORT='47711'
  function Boot([string]$tag){
    $o=Join-Path $base "out-$tag.log"; $e=Join-Path $base "err-$tag.log"
    $p=Start-Process -FilePath 'node' -ArgumentList (Join-Path $root.FullName 'server\index.js') -WorkingDirectory $root.FullName -RedirectStandardOutput $o -RedirectStandardError $e -PassThru
    $script:gwPids+=$p.Id
    $ok=$false
    for($i=0;$i -lt 40;$i++){Start-Sleep -Milliseconds 500
      if((Test-Path $o) -and ((Get-Content $o -Raw -ErrorAction SilentlyContinue) -match 'server up')){$ok=$true;break}
      if($p.HasExited){break}}
    return @{P=$p;Ok=$ok;Out=$o;Err=$e}
  }
  $b1=Boot 'first'
  if(-not $b1.Ok){Say (Get-Content $b1.Out,$b1.Err -Raw -ErrorAction SilentlyContinue);throw 'boot 1 failed'}
  $devFile=Join-Path $data 'device.json'
  if(-not (Test-Path $devFile)){throw 'device.json missing'}
  $dev1=(Get-Content $devFile -Raw | ConvertFrom-Json -EA Stop).id
  Say ("PASS 3: first boot up + identity minted (device {0})" -f $dev1)
  $mySid=[System.Security.Principal.WindowsIdentity]::GetCurrent().User.Value
  function Assert-OwnerOnlyAcl($p,$isDir){
    $a=Get-Acl $p
    $rules=@($a.Access)
    if(-not $a.AreAccessRulesProtected){throw ('ACL not protected (inheritance on): '+$p)}
    if($rules.Count -ne 1){throw ('ACL rule count != 1 on '+$p+': '+$rules.Count)}
    $ruleSid=$rules[0].IdentityReference.Translate([System.Security.Principal.SecurityIdentifier]).Value
    if($ruleSid -ne $mySid){throw ('ACL rule SID is not the current user on '+$p+': '+$ruleSid)}
    $rw=[string]$rules[0].FileSystemRights
    if($isDir){ if($rw -notmatch 'FullControl'){throw ('ACL rights not FullControl on dir '+$p+': '+$rw)} }
    elseif($rw -notmatch 'FullControl' -and ($rw -notmatch 'Read' -or $rw -notmatch 'Write')){throw ('ACL rights missing Read/Write on file '+$p+': '+$rw)} # files: the code grants owner R+W by design (least privilege) - FullControl is NOT required for privacy
    if($rules[0].AccessControlType -ne 'Allow'){throw ('ACL rule is not Allow on '+$p)}
  }
  Assert-OwnerOnlyAcl $devFile $false
  Assert-OwnerOnlyAcl $data $true
  Say 'PASS 4: device identity ACL is owner-only (exact SID, file AND directory)'
  $code403=$false
  try{Invoke-WebRequest -Uri 'http://127.0.0.1:47711/health' -UseBasicParsing | Out-Null}
  catch{ if($_.Exception.Response -and [int]$_.Exception.Response.StatusCode -eq 403){$code403=$true} }
  $r=Invoke-WebRequest -Uri 'http://127.0.0.1:47711/health?nonce=boottest1234' -Headers @{'X-GW-Key'='boot-test-token-123'} -UseBasicParsing
  $h=$r.Content | ConvertFrom-Json -EA Stop
  if(-not ($code403 -and $h.ok -eq $true -and $h.deviceId -eq $dev1 -and $h.sig)){throw 'health gate wrong'}
  # verify the health signature CRYPTOGRAPHICALLY against the minted pubkey (deviceId+LF+nonce)
  $verifyFile=Join-Path $base 'verify-sig.js'
  Set-Content -Path $verifyFile -Encoding ASCII -Value "const c=require('crypto');const[pub,dev,nonce,sig]=process.argv.slice(2);const ok=c.verify(null,Buffer.from(dev+String.fromCharCode(10)+nonce,'utf8'),c.createPublicKey({key:Buffer.from(pub,'base64'),format:'der',type:'spki'}),Buffer.from(sig,'hex'));process.exit(ok?0:1)"
  $devJson=(Get-Content $devFile -Raw | ConvertFrom-Json -EA Stop)
  & node $verifyFile $devJson.pub $dev1 'boottest1234' $h.sig
  if($LASTEXITCODE -ne 0){throw 'health signature did NOT verify against the minted key'}
  Say 'PASS 5: /health without key -> 403, with key -> 200, signature verified'
  Stop-Process -Id $b1.P.Id -Force -ErrorAction SilentlyContinue
  $b2=Boot 'second'
  if(-not $b2.Ok){throw 'boot 2 failed'}
  $devJ2=(Get-Content $devFile -Raw | ConvertFrom-Json -EA Stop)
  $dev2=$devJ2.id
  if($dev2 -eq $dev1 -and $devJ2.pub -eq $devJson.pub -and $devJ2.priv -eq $devJson.priv){Say 'PASS 6: second boot - identical identity (full keypair equality)'}else{throw 'identity changed across boots'}
  Stop-Process -Id $b2.P.Id -Force -ErrorAction SilentlyContinue
  Add-Content -Path $devFile -Value 'corrupt'
  # third boot via cmd (no PS error-stream serialization noise): stderr/stdout as raw files
  $e3=Join-Path $base 'err-third.log'; $o3=Join-Path $base 'out-third.log'
  $thirdJs=Join-Path $root.FullName 'server\index.js'
  cmd /c "node `"$thirdJs`" 1> `"$o3`" 2> `"$e3`"" | Out-Null
  $exit3=$LASTEXITCODE
  $errTxt=(Get-Content $e3 -Raw -ErrorAction SilentlyContinue)
  $outTxt=(Get-Content $o3 -Raw -ErrorAction SilentlyContinue)
  Say ($errTxt.Trim())
  # fail-loud attributed to the IDENTITY check: nonzero exit AND no listener AND the refusal names
  # the identity failure (corrupt JSON -> SyntaxError from the device load; pair mismatch ->
  # 'refusing to boot'). A generic crash elsewhere does NOT pass.
  # the REAL contract: nonzero exit AND no listener AND the corrupt identity NOT rewritten.
  # Stream text is only attribution evidence (the ACL helper's powershell children can pollute
  # stderr with CLIXML progress records, so exact stream matching is not the contract); the pinned
  # b78d70e raw SyntaxError and the wip clean 'refusing to boot' are BOTH acceptable refusals.
  $attributed=($errTxt -match 'refusing to boot') -or ($errTxt -match 'SyntaxError') -or ($outTxt -match 'refusing to boot') -or ($outTxt -match 'SyntaxError')
  $listener=$false;try{$listener=[bool](Get-NetTCPConnection -LocalPort 47711 -State Listen -ErrorAction SilentlyContinue | Select-Object -First 1)}catch{}
  $devAfter=(Get-Content $devFile -Raw -ErrorAction SilentlyContinue)
  $notRewritten=($devAfter -match 'corrupt') # the identity must stay EXACTLY as corrupted - a silent re-mint is the failure this test hunts
  if($exit3 -ne 0 -and (-not $listener) -and $notRewritten -and $attributed){Say ('PASS 7: corrupt identity -> boot refused (fail-loud, exit '+$exit3+', identity not rewritten)')}
  else{$errLen=0;$outLen=0;if($errTxt){$errLen=$errTxt.Length};if($outTxt){$outLen=$outTxt.Length}
   $errHead='';$outHead='';if($errTxt){$errHead=$errTxt.Substring(0,[Math]::Min(160,$errLen))};if($outTxt){$outHead=$outTxt.Substring(0,[Math]::Min(160,$outLen))}
   throw ('corrupt identity boot not refused: exit='+$exit3+' listener='+$listener+' notRewritten='+$notRewritten+' attributed='+$attributed+' errBytes='+$errLen+' outBytes='+$outLen+' errHead=['+$errHead+'] outHead=['+$outHead+']')}
  Say 'DONE - all checks passed'
}catch{
  Say ('FAIL: '+$_.Exception.Message)
  Say ($_.ScriptStackTrace)
  Say 'send this whole output back + boot-test-log.txt'
}finally{
  foreach($gp in $script:gwPids){Stop-Process -Id $gp -Force -ErrorAction SilentlyContinue}
  Remove-Item -Recurse -Force $base -ErrorAction SilentlyContinue
  try{Stop-Transcript | Out-Null}catch{}
  Read-Host 'Press Enter to close'
}
